def parse_response(protocol,stream,raw):
 if not stream:

  if obj.get('error'):raise ValueError('response error: '+json.dumps(obj['error']))
  return obj
 events=[];done=False
 for line in raw.splitlines():
  if not line.startswith('data:'):continue
  data=line[5:].strip()
  if data=='[DONE]':done=True;continue
  event=json.loads(data)
  if event.get('error') or event.get('type') in ['error','response.failed']:
   raise ValueError('stream error: '+json.dumps(event))
  events.append(event)
 if protocol=='chat':
  assert done,'Chat stream missing [DONE]'
  msg={'role':'assistant','content':None};calls={};finish=None;usage=None
  for event in events:
   usage=event.get('usage') or usage
   for choice in event.get('choices',[]):
    assert choice.get('index',0)==0,'unexpected multiple choices'
    delta=choice.get('delta') or {};finish=choice.get('finish_reason') or finish
    for field in ['content','reasoning_content','refusal']:
     if isinstance(delta.get(field),str):msg[field]=(msg.get(field) or '')+delta[field]
    for part in delta.get('tool_calls',[]):
     call=calls.setdefault(part['index'],{'id':'','type':'function','function':{'name':'','arguments':''}})
     for field in ['id']:
      if part.get(field):call[field]+=part[field]
     fn=part.get('function') or {}
     for field in ['name','arguments']:
      if fn.get(field):call['function'][field]+=fn[field]
  if calls:msg['tool_calls']=[calls[i] for i in sorted(calls)]
  return {'choices':[{'message':msg,'finish_reason':finish}],'usage':usage}
 if protocol=='responses':
  terminal=next((e for e in reversed(events) if e.get('type') in ['response.completed','response.incomplete']),None)
  assert terminal,'Responses stream missing terminal response'
  seq=[e['sequence_number'] for e in events if 'sequence_number' in e]

  return terminal['response']

  elif kind=='content_block_start':blocks[event['index']]=copy.deepcopy(event['content_block'])

 obj['content']=[blocks[i] for i in sorted(blocks)]
