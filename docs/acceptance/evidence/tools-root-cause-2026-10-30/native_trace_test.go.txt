type causeTrace struct {
 Name string `json:"name"`
 Prepared json.RawMessage `json:"prepared_chat_request"`
 NativeRequest json.RawMessage `json:"native_request"`
 NativeResponse string `json:"native_response"`
 Chunks []json.RawMessage `json:"normalized_chunks"`
 ClientStatus int `json:"client_status"`
 ClientResponse string `json:"client_response"`
 HTTPCalls int `json:"http_calls"`
 Seconds float64 `json:"seconds"`
}
type causeBody struct {io.ReadCloser; buf *bytes.Buffer}
func(b *causeBody)Read(p []byte)(int,error){n,e:=b.ReadCloser.Read(p);b.buf.Write(p[:n]);return n,e}
type causeAdapter struct {provider.Adapter; record *causeTrace}
func(a *causeAdapter)Stream(ctx context.Context,raw json.RawMessage,input provider.Request,emit func([]byte)error)error{
 a.record.Prepared=append([]byte(nil),input.Body...)
 return a.Adapter.Stream(ctx,raw,input,func(b []byte)error{a.record.Chunks=append(a.record.Chunks,append([]byte(nil),b...));return emit(b)})
}
