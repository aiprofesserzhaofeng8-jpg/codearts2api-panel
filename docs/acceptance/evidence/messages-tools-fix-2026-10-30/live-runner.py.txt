def parse_response(protocol,stream,raw):
 if not stream:
  obj=json.loads(raw)
  if obj.get('error'):raise ValueError('response error: '+json.dumps(obj['error']))
  return obj
 events=[];done=False
 for line in raw.splitlines():
  if not line.startswith('data:'):continue
  data=line[5:].strip()
  if data=='[DONE]':done=True;continue
  event=json.loads(data)
  if event.get('error') or event.get('type') in ['error','response.failed']:
   raise ValueError('stream error: '+json.dumps(event))
  events.append(event)
 if protocol=='chat':
  assert done,'Chat stream missing [DONE]'
  msg={'role':'assistant','content':None};calls={};finish=None;usage=None
  for event in events:
   usage=event.get('usage') or usage
   for choice in event.get('choices',[]):
    assert choice.get('index',0)==0,'unexpected multiple choices'
    delta=choice.get('delta') or {};finish=choice.get('finish_reason') or finish
    for field in ['content','reasoning_content','refusal']:
     if isinstance(delta.get(field),str):msg[field]=(msg.get(field) or '')+delta[field]
    for part in delta.get('tool_calls',[]):
     call=calls.setdefault(part['index'],{'id':'','type':'function','function':{'name':'','arguments':''}})
     for field in ['id']:
      if part.get(field):call[field]+=part[field]
     fn=part.get('function') or {}
     for field in ['name','arguments']:
      if fn.get(field):call['function'][field]+=fn[field]
  if calls:msg['tool_calls']=[calls[i] for i in sorted(calls)]
  return {'choices':[{'message':msg,'finish_reason':finish}],'usage':usage}
 if protocol=='responses':
  terminal=next((e for e in reversed(events) if e.get('type') in ['response.completed','response.incomplete']),None)
  assert terminal,'Responses stream missing terminal response'
  seq=[e['sequence_number'] for e in events if 'sequence_number' in e]
  assert seq==sorted(set(seq)),'Responses event sequence invalid'
  return terminal['response']
 assert events and events[-1].get('type')=='message_stop','Messages stream missing message_stop'
 obj={};blocks={};fragments={}
 for event in events:
  kind=event['type']
  if kind=='message_start':obj=copy.deepcopy(event['message'])
  elif kind=='content_block_start':blocks[event['index']]=copy.deepcopy(event['content_block'])
  elif kind=='content_block_delta':
   i=event['index'];delta=event['delta'];kind_delta=delta['type']
   if kind_delta=='input_json_delta':fragments[i]=fragments.get(i,'')+delta['partial_json']
   else:
    field={'text_delta':'text','thinking_delta':'thinking','signature_delta':'signature'}.get(kind_delta)
    assert field,'unhandled Messages delta: '+kind_delta
    blocks[i][field]=blocks[i].get(field,'')+delta[field]
  elif kind=='message_delta':obj.update(event.get('delta',{}));obj.setdefault('usage',{}).update(event.get('usage',{}))
 for i,fragment in fragments.items():blocks[i]['input']=json.loads(fragment)
 obj['content']=[blocks[i] for i in sorted(blocks)]
 return obj

def make_request(protocol,model,stream,prompt):
 common={'model':model,'stream':stream,'thinking':{'type':'disabled'}}
 if protocol=='chat':
  common.update(messages=[{'role':'user','content':prompt}],tools=[{'type':'function','function':{'name':tool_name,'description':description,'parameters':schema}}],max_tokens=640)
  if stream:common['stream_options']={'include_usage':True}
  return '/v1/chat/completions',common
 if protocol=='responses':
  common.update(input=[{'role':'user','content':prompt}],tools=[{'type':'function','name':tool_name,'description':description,'parameters':schema}],max_output_tokens=640,store=False)
  return '/v1/responses',common
 common.update(messages=[{'role':'user','content':prompt}],tools=[{'name':tool_name,'description':description,'input_schema':schema}],max_tokens=640)
 return '/v1/messages',common

def extract(protocol,obj):
 if protocol=='chat':
  choice=obj['choices'][0];msg=choice['message'];finish=choice.get('finish_reason')
  calls=[{'id':x['id'],'name':x['function']['name'],'args':json.loads(x['function']['arguments'])} for x in msg.get('tool_calls',[])]
  answer=msg.get('content') or '';terminal='tool_calls' if calls else 'stop'
 elif protocol=='responses':
  calls=[{'id':x['call_id'],'name':x['name'],'args':json.loads(x['arguments'])} for x in obj['output'] if x.get('type')=='function_call']
  answer=''.join(p.get('text','') for x in obj['output'] for p in x.get('content',[]) if p.get('type')=='output_text');finish=obj['status'];terminal='completed'
 else:
  calls=[{'id':x['id'],'name':x['name'],'args':x['input']} for x in obj['content'] if x.get('type')=='tool_use']
  answer=''.join(x.get('text','') for x in obj['content'] if x.get('type')=='text');finish=obj.get('stop_reason');terminal='tool_use' if calls else 'end_turn'
 assert finish==terminal,'bad finish '+str(finish)
 assert len({c['id'] for c in calls})==len(calls),'duplicate call IDs within one turn'
 for c in calls:
  assert c['id'] and c['name']==tool_name and isinstance(c['args'],dict) and set(c['args'])=={'path'},'invalid tool call shape'
  assert isinstance(c['args']['path'],str),'path must be a string'
 return calls,answer

def append_results(protocol,body,obj,results):
 if protocol=='chat':body['messages'] += [obj['choices'][0]['message']]+[{'role':'tool','tool_call_id':c['id'],'content':text} for c,text,error in results]
 elif protocol=='responses':body['input'] += obj['output']+[{'type':'function_call_output','call_id':c['id'],'output':text} for c,text,error in results]
 else:body['messages'] += [{'role':'assistant','content':obj['content']},{'role':'user','content':[{'type':'tool_result','tool_use_id':c['id'],'content':text,**({'is_error':True} if error else {})} for c,text,error in results]}]
